**Goal:** 交付 DSH for VS Code v0.3.0：SSH Remote 隧道支持、编辑器右上角图标、对话框自由上传图片（非视觉模型自动降级）、dsh 新版默认弹浏览器修复。

export type UriLike = { toString(): string };
export function isRemoteName(name: string | undefined): boolean {
  return typeof name === 'string' && name.trim() !== '';
}
export type ExternalUriApp = { asExternalUri(uri: UriLike): Promise<UriLike> };
export function createUrlResolver(app: ExternalUriApp) {
  return async (url: string): Promise<string> => {
    try {
      const r = await app.asExternalUri({ toString: () => url } as UriLike);
      return r.toString();
    } catch {
      return url; // 隧道失败：退回原 URL（本地可达场景仍可用）
    }
  };
}
